cmake_minimum_required(VERSION 3.20)
project(KernelForge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kernelforge src/compiler.cpp)
target_include_directories(kernelforge PUBLIC include)
target_compile_options(kernelforge PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kforge src/main.cpp)
target_link_libraries(kforge PRIVATE kernelforge)
include(CTest)
add_executable(compiler_tests tests/compiler_tests.cpp)
target_link_libraries(compiler_tests PRIVATE kernelforge)
add_test(NAME compiler COMMAND compiler_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME generated_python COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/python/check_codegen.py $<TARGET_FILE:kforge>)

add_test(NAME distributed_inference_contracts
 COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_distributed.py -v)
set_tests_properties(distributed_inference_contracts PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
